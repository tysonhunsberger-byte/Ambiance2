class AmbianceMainWindow(QMainWindow):
    """Minimal main window to keep the project runnable while we restore features."""

    def __init__(self) -> None:
        super().__init__()
        self.setWindowTitle("Ambiance – Improved UI (Clean Shell)")
        self.resize(1000, 700)

        root = QWidget(self)
        self.setCentralWidget(root)
        layout = QVBoxLayout(root)

        hdr = QLabel("Ambiance – Plugin Rack")
        hdr.setStyleSheet("font-size:20px; font-weight:600; margin:12px 0;")
        layout.addWidget(hdr)

        # Plugin Rack block
        if PluginRackManager is None:
            warn = QLabel(
                "Plugin rack unavailable: could not import PluginRackManager.\n\n"
                f"Import error: {globals().get('_PLUGINS_IMPORT_ERROR','unknown')}"
            )
            warn.setWordWrap(True)
            layout.addWidget(warn)
        else:
            rack = PluginRackWidget(self)
            layout.addWidget(rack)

        # Optional note about WebEngine status
        if not WEBENGINE_AVAILABLE:
            info = QLabel(
                "Qt WebEngine is not available. The embedded browser UI can be enabled by installing PyQt6-WebEngine."
            )
            info.setWordWrap(True)
            info.setStyleSheet("color:#888; font-size:12px; margin-top:8px;")
            layout.addWidget(info)


def main(argv: Optional[list[str]] = None) -> int:
    logging.basicConfig(level=logging.INFO)
    app = QApplication(sys.argv if argv is None else argv)
    win = AmbianceMainWindow()
    win.show()
    return app.exec()


if __name__ == "__main__":  # pragma: no cover
    raise SystemExit(main())


class PluginRackWidget(QFrame):
    """Minimal Plugin Rack UI backed by PluginRackManager.

    - Shows workspace and discovered plugins
    - A/B lanes (for stream 'Main') with add/remove/toggle
    """

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self.setFrameShape(QFrame.StyledPanel)
        self.setStyleSheet(
            "QFrame { border:1px solid #444; border-radius:8px; padding:12px; background:#1e1e1e; }"
        )
        self.setSizePolicy(QSizePolicy.Policy.Expanding, QSizePolicy.Policy.Preferred)

        self.manager = PluginRackManager()  # type: ignore[arg-type]

        outer = QVBoxLayout(self)

        # Header row with workspace path and controls
        hdr_row = QHBoxLayout()
        self.workspace_label = QLabel()
        self.workspace_label.setTextInteractionFlags(Qt.TextSelectableByMouse)
        hdr_row.addWidget(self.workspace_label, 1)

        self.refresh_btn = QPushButton("Refresh")
        self.refresh_btn.clicked.connect(self.refresh)
        hdr_row.addWidget(self.refresh_btn)

        self.toggle_btn = QPushButton("Toggle A/B")
        self.toggle_btn.clicked.connect(self.toggle_lane)
        hdr_row.addWidget(self.toggle_btn)

        outer.addLayout(hdr_row)

        # Body: plugin list (left) and lanes (right)
        body = QHBoxLayout()
        outer.addLayout(body)

        # Discovered plugins list
        left_box = QVBoxLayout()
        body.addLayout(left_box, 1)
        left_box.addWidget(QLabel("Discovered Plugins"))
        self.plugins_list = QListWidget()
        left_box.addWidget(self.plugins_list, 1)

        add_row = QHBoxLayout()
        self.add_a_btn = QPushButton("Add → Lane A")
        self.add_b_btn = QPushButton("Add → Lane B")
        self.add_a_btn.clicked.connect(lambda: self.add_selected_to_lane("A"))
        self.add_b_btn.clicked.connect(lambda: self.add_selected_to_lane("B"))
        add_row.addWidget(self.add_a_btn)
        add_row.addWidget(self.add_b_btn)
        left_box.addLayout(add_row)

        # Lanes panel
        right_box = QVBoxLayout()
        body.addLayout(right_box, 1)

        lanes_row = QHBoxLayout()
        right_box.addLayout(lanes_row, 1)

        self.lane_a = self._make_lane_box("Lane A")
        self.lane_b = self._make_lane_box("Lane B")
        lanes_row.addWidget(self.lane_a["group"], 1)
        lanes_row.addWidget(self.lane_b["group"], 1)

        # Initial population
        self.refresh()

    def _make_lane_box(self, title: str) -> dict[str, object]:
        grp = QGroupBox(title)
        grp.setStyleSheet("QGroupBox{ font-weight:600; }")
        v = QVBoxLayout(grp)
        lst = QListWidget()
        v.addWidget(lst, 1)
        btn_row = QHBoxLayout()
        remove_btn = QPushButton("Remove Selected")
        btn_row.addWidget(remove_btn)
        v.addLayout(btn_row)
        lane_key = "A" if "A" in title.upper() else "B"
        remove_btn.clicked.connect(lambda: self.remove_selected_from_lane(lane_key))
        return {"group": grp, "list": lst, "remove_btn": remove_btn, "key": lane_key}

    # --- Data ops
    def refresh(self) -> None:
        status = self.manager.status()  # type: ignore[union-attr]
        self.workspace_label.setText(f"Workspace: {status.get('workspace','(unknown)')}")

        # Populate plugin list
        self.plugins_list.clear()
        for item in status.get("plugins", []):
            name = item.get("name") or Path(item.get("path","?")).name
            fmt = item.get("format", "?")
            path = item.get("path", "?")
            li = QListWidgetItem(f"{name} [{fmt}]\n{path}")
            li.setData(Qt.UserRole, item)
            self.plugins_list.addItem(li)

        # Populate lanes
        streams = status.get("streams", {})
        main = streams.get("Main", {"lanes": {"A": [], "B": []}})
        for lane_dict, lane_state in ((self.lane_a, main["lanes"].get("A", [])), (self.lane_b, main["lanes"].get("B", []))):
            lst: QListWidget = lane_dict["list"]  # type: ignore[index]
            lst.clear()
            for entry in lane_state:
                slot = entry.get("slot")
                plugin = entry.get("plugin", {})
                name = plugin.get("name") or Path(plugin.get("path","?")).name
                fmt = plugin.get("format", "?")
                text = f"[{slot}] {name} ({fmt})"
                li = QListWidgetItem(text)
                li.setData(Qt.UserRole, entry)
                lst.addItem(li)

    def add_selected_to_lane(self, lane: str) -> None:
        item = self.plugins_list.currentItem()
        if not item:
            return
        data = item.data(Qt.UserRole) or {}
        path = data.get("path")
        if not path:
            return
        try:
            self.manager.add_plugin(stream="Main", lane=lane, path=str(path))  # type: ignore[arg-type]
        except Exception:
            pass
        self.refresh()

    def remove_selected_from_lane(self, lane: str) -> None:
        lane_dict = self.lane_a if lane.upper() == "A" else self.lane_b
        lst: QListWidget = lane_dict["list"]  # type: ignore[index]
        item = lst.currentItem()
        if not item:
            return
        entry = item.data(Qt.UserRole) or {}
        slot = entry.get("slot")
        path = None
        try:
            self.manager.remove_plugin(stream="Main", lane=lane, slot=slot, path=path)  # type: ignore[arg-type]
        except Exception:
            pass
        self.refresh()

    def toggle_lane(self) -> None:
        try:
            self.manager.toggle_lane("Main")  # type: ignore[arg-type]
        except Exception:
            pass
        self.refresh()
